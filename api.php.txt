<?php

header("Content-Type: application/json; charset=UTF-8");

if ($_SERVER["REQUEST_METHOD"] !== "POST") {
    echo json_encode([
        "success" => false,
        "message" => "يسمح بطلب POST فقط"
    ], JSON_UNESCAPED_UNICODE);
    exit;
}

/*
|--------------------------------------------------------------------------
| رابط Google Apps Script
|--------------------------------------------------------------------------
*/

$googleScriptUrl = "https://script.google.com/macros/s/AKfycbz_-zVzd_fEkCtKUE_f8fU7uX6dXMIliBt_pdjBlN3XU0UVG-PlrvM9sKGRbarp33B8/exec";


/*
|--------------------------------------------------------------------------
| قراءة بيانات الاستمارة
|--------------------------------------------------------------------------
*/

$data = $_POST;


/*
|--------------------------------------------------------------------------
| التحقق
|--------------------------------------------------------------------------
*/

if (empty($data)) {

    $json = file_get_contents("php://input");

    if (!empty($json)) {
        $data = json_decode($json, true);
    }
}

if (empty($data)) {

    echo json_encode([
        "success" => false,
        "message" => "لم تصل بيانات الاستمارة"
    ], JSON_UNESCAPED_UNICODE);

    exit;
}


/*
|--------------------------------------------------------------------------
| إرسال البيانات إلى Google Apps Script
|--------------------------------------------------------------------------
*/

$ch = curl_init($googleScriptUrl);

curl_setopt($ch, CURLOPT_POST, true);

curl_setopt(
    $ch,
    CURLOPT_POSTFIELDS,
    http_build_query($data)
);

curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);

curl_setopt($ch, CURLOPT_FOLLOWLOCATION, true);

curl_setopt($ch, CURLOPT_TIMEOUT, 30);

curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, true);


$response = curl_exec($ch);

$error = curl_error($ch);

$httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);

curl_close($ch);


/*
|--------------------------------------------------------------------------
| فحص النتيجة
|--------------------------------------------------------------------------
*/

if ($error) {

    echo json_encode([
        "success" => false,
        "message" => "خطأ في الاتصال: " . $error
    ], JSON_UNESCAPED_UNICODE);

    exit;
}


if ($httpCode >= 200 && $httpCode < 400) {

    echo json_encode([
        "success" => true,
        "message" => "تم إرسال البيانات إلى جدول Google Sheets بنجاح"
    ], JSON_UNESCAPED_UNICODE);

    exit;
}


echo json_encode([
    "success" => false,
    "message" => "لم يتمكن الخادم من إرسال البيانات إلى Google Sheets",
    "http_code" => $httpCode
], JSON_UNESCAPED_UNICODE);

?>

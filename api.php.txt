document.getElementById("myForm").addEventListener("submit", async function(e) {

    e.preventDefault();

    updateDateTime();

    const form = e.target;
    const button = document.getElementById("submitBtn");

    button.disabled = true;
    button.textContent = "⏳ جاري الحفظ...";

    try {

        // إرسال الاستمارة إلى api.php
        const formData = new FormData(form);

        const response = await fetch("api.php", {
            method: "POST",
            body: formData
        });

        // قراءة رد api.php
        const result = await response.json();

        console.log("رد الخادم:", result);

        if (result.success === true) {

            alert("✅ " + result.message);

            // زيادة التسلسل
            serial++;

            localStorage.setItem(
                "serial",
                serial
            );

            // تفريغ الاستمارة
            form.reset();

            // تحديث التسلسل
            document.getElementById("serial")
                .textContent = serial;

            document.getElementById("hidden_serial")
                .value = serial;

            // تحديث التاريخ والوقت
            updateDateTime();

        } else {

            alert(
                "❌ لم يتم حفظ البيانات:\n\n" +
                result.message
            );
        }

    } catch (error) {

        console.error("خطأ:", error);

        alert(
            "❌ حدث خطأ في الاتصال بـ api.php\n\n" +
            error.message
        );

    } finally {

        button.disabled = false;

        button.textContent =
            "📤 إرسال الاستمارة";
    }

});
